-- PATCH 1 : demande de réservation sécurisée. À exécuter APRÈS schema.sql (SQL Editor > Run). NON TESTÉ. alter table bookings add column if not exists pickup_text text; create or replace function request_booking( p_vehicle_name text, p_start date, p_end date, p_name text, p_phone text, p_location text ) returns jsonb language plpgsql security definer set search_path=public as $$ declare v vehicles; s timestamptz; e timestamptz; min_d int := (select value::int from settings where key='min_days'); hold_m int := (select value::int from settings where key='hold_minutes'); days int := p_end - p_start; sub numeric; bid uuid; begin perform expire_holds(); p_phone := regexp_replace(coalesce(p_phone,''), '\s', '', 'g'); if length(trim(coalesce(p_name,''))) < 2 or length(p_name) > 80 then return jsonb_build_object('ok',false,'error','name'); end if; if p_phone !~ '^(\+216)?[2459][0-9]{7}$' then return jsonb_build_object('ok',false,'error','phone'); end if; if p_start is null or p_end is null or p_start < current_date or days < min_d then return jsonb_build_object('ok',false,'error','dates'); end if; -- anti-spam : max 5 demandes en attente par téléphone et par heure if (select count(*) from bookings where guest_phone = p_phone and created_at > now() - interval '1 hour') >= 5 then return jsonb_build_object('ok',false,'error','too_many'); end if; s := (p_start::timestamp) at time zone 'Africa/Tunis'; e := (p_end::timestamp) at time zone 'Africa/Tunis'; -- première unité libre de ce modèle (gère plusieurs voitures du même modèle) select * into v from vehicles ve where ve.name = p_vehicle_name and ve.status = 'active' and not exists (select 1 from bookings b where b.vehicle_id = ve.id and b.status in ('pending','confirmed','cancellation_requested','blocked') and tstzrange(b.start_at,b.end_at) && tstzrange(s,e)) order by ve.sort limit 1; if not found then return jsonb_build_object('ok',false,'error','unavailable'); end if; -- prix HTVA recalculé ICI (jamais depuis le navigateur) : tarif spécial de la date sinon tarif de base if v.daily_price_htva is not null then select sum(coalesce((select r.daily_price_htva from price_rules r where r.vehicle_id = v.id and d::date between r.starts_on and r.ends_on order by r.created_at desc limit 1), v.daily_price_htva)) into sub from generate_series(p_start, p_end - 1, interval '1 day') d; end if; begin insert into bookings(guest_name,guest_phone,vehicle_id,start_at,end_at,status,hold_expires_at,pickup_text,quote) values (trim(p_name), p_phone, v.id, s, e, 'pending', now() + make_interval(mins => hold_m), left(p_location,80), jsonb_build_object('days',days,'subtotal_htva',sub,'vat_rate',(select value from settings where key='vat_rate'), 'note','estimation HTVA, taxes à confirmer')) returning id into bid; exception when exclusion_violation then return jsonb_build_object('ok',false,'error','unavailable'); end; insert into audit_log(action,entity,entity_id,detail) values ('booking_requested','booking',bid::text,jsonb_build_object('vehicle',v.name)); return jsonb_build_object('ok',true,'id',bid,'vehicle',v.name,'days',days,'subtotal_htva',sub,'hold_minutes',hold_m); end $$; grant execute on function request_booking(text,date,date,text,text,text) to anon, authenticated; grant execute on function expire_holds() to anon, authenticated; -- Libération automatique chaque minute (si pg_cron n'est pas activé, l'étape est ignorée : le panneau admin libère aussi à l'ouverture) do $$ begin create extension if not exists pg_cron; perform cron.schedule('expire-holds','* * * * *','select public.expire_holds()'); exception when others then raise notice 'pg_cron non activé : %', sqlerrm; end $$;
